# ui-render: 2D painter (CPU only) and the Vulkan renderer that draws its output.
#
# Targets:
#   ui-render-paint  (r1ui::render_paint)  Painter, clip stack, SDF reference. No Vulkan or OS
#                    dependency, so it always builds and its unit tests run in the fast tier.
#   ui-render        (r1ui::render)        Device, window/offscreen targets, textures, pipelines and
#                    the legacy Renderer facade. Needs the Vulkan SDK (headers, loader, glslc); when
#                    it is missing the module is skipped with a warning instead of failing configure.
#
# Shaders: source/ui-render/shaders/*.vert|*.frag are compiled with glslc at build time and the
# SPIR-V is embedded into ui-render as generated byte arrays (shaders/embed_spirv.cmake), so the
# executable has no runtime shader files to lose.

add_library(ui-render-paint STATIC paint/Painter.cpp paint/ClipStack.cpp paint/SdfReference.cpp)
add_library(r1ui::render_paint ALIAS ui-render-paint)
target_include_directories(ui-render-paint PUBLIC include)
target_link_libraries(ui-render-paint PUBLIC r1ui::core PRIVATE r1ui_options)

find_package(Vulkan)
if(NOT Vulkan_FOUND)
  message(WARNING "Vulkan SDK not found (set VULKAN_SDK): ui-render and the preview are disabled")
elseif(NOT Vulkan_GLSLC_EXECUTABLE)
  message(WARNING "glslc not found in the Vulkan SDK: ui-render and the preview are disabled")
else()
  # Compile every shader and generate one .cpp per SPIR-V blob (symbols declared in vulkan/Shaders.h).
  set(R1UI_SHADER_DIR ${CMAKE_CURRENT_SOURCE_DIR}/shaders)
  set(R1UI_SHADER_GEN_DIR ${CMAKE_CURRENT_BINARY_DIR}/generated)
  file(MAKE_DIRECTORY ${R1UI_SHADER_GEN_DIR})
  set(R1UI_SHADER_SOURCES "")
  foreach(entry
      "sdf.vert|kSdfVertSpv" "sdf.frag|kSdfFragSpv"
      "textured.vert|kTexturedVertSpv" "textured.frag|kTexturedFragSpv")
    string(REPLACE "|" ";" parts "${entry}")
    list(GET parts 0 shader)
    list(GET parts 1 symbol)
    set(spv ${R1UI_SHADER_GEN_DIR}/${shader}.spv)
    set(gen ${R1UI_SHADER_GEN_DIR}/${shader}.spv.cpp)
    add_custom_command(
      OUTPUT ${spv}
      COMMAND ${Vulkan_GLSLC_EXECUTABLE} -O --target-env=vulkan1.2 -o ${spv} ${R1UI_SHADER_DIR}/${shader}
      DEPENDS ${R1UI_SHADER_DIR}/${shader}
      COMMENT "glslc ${shader}"
      VERBATIM)
    add_custom_command(
      OUTPUT ${gen}
      COMMAND ${CMAKE_COMMAND} -DINPUT=${spv} -DOUTPUT=${gen} -DSYMBOL=${symbol}
              -P ${R1UI_SHADER_DIR}/embed_spirv.cmake
      DEPENDS ${spv} ${R1UI_SHADER_DIR}/embed_spirv.cmake
      COMMENT "Embedding ${shader}.spv"
      VERBATIM)
    list(APPEND R1UI_SHADER_SOURCES ${gen})
  endforeach()

  add_library(ui-render STATIC
    vulkan/GpuResources.cpp
    vulkan/RenderDevice.cpp
    vulkan/Pipelines.cpp
    vulkan/UploadQueue.cpp
    vulkan/FrameRecorder.cpp
    vulkan/Texture.cpp
    vulkan/WindowTarget.cpp
    vulkan/OffscreenTarget.cpp
    vulkan/Renderer.cpp
    ${R1UI_SHADER_SOURCES})
  add_library(r1ui::render ALIAS ui-render)
  target_include_directories(ui-render PUBLIC include PRIVATE vulkan)
  target_compile_definitions(ui-render PRIVATE VK_USE_PLATFORM_WIN32_KHR)
  target_link_libraries(ui-render PUBLIC r1ui::platform r1ui::render_paint PRIVATE Vulkan::Vulkan r1ui_options)
endif()

# yet; once it does, drop this block. Placed last so the GPU tests can see the ui-render target.
if(BUILD_TESTING)

endif()
